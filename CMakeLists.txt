cmake_minimum_required(VERSION 3.25)
project(ServiceStarter VERSION 1.0.0 LANGUAGES CXX)

# Standalone library. Add with add_subdirectory(third_party/service_starter) and link service_starter::service_starter.
option(SERVICE_STARTER_BUILD_TESTS "Build service_starter tests" ${PROJECT_IS_TOP_LEVEL})
option(SERVICE_STARTER_STRICT_WARNINGS "Treat warnings as errors in service_starter" ON)
option(SERVICE_STARTER_BUILD_EXAMPLES "Build the service_starter example executable" ${PROJECT_IS_TOP_LEVEL})

find_package(Threads REQUIRED)

add_library(service_starter STATIC src/host.cpp)
add_library(service_starter::service_starter ALIAS service_starter)
if(WIN32)
    target_sources(service_starter PRIVATE src/host_windows.cpp)
    target_link_libraries(service_starter PRIVATE advapi32)
    target_compile_definitions(service_starter PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
else()
    target_sources(service_starter PRIVATE src/host_posix.cpp)
endif()
target_include_directories(service_starter PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(service_starter PUBLIC cxx_std_20)
target_link_libraries(service_starter PUBLIC Threads::Threads)

function(service_starter_configure target)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8 /EHsc)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(SERVICE_STARTER_STRICT_WARNINGS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(SERVICE_STARTER_STRICT_WARNINGS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
service_starter_configure(service_starter)

if(SERVICE_STARTER_BUILD_EXAMPLES)
    add_executable(service_starter_example examples/example_service.cpp)
    target_link_libraries(service_starter_example PRIVATE service_starter::service_starter)
    if(WIN32)
        target_compile_definitions(service_starter_example PRIVATE UNICODE _UNICODE)
    endif()
    service_starter_configure(service_starter_example)
endif()

if(SERVICE_STARTER_BUILD_TESTS)
    include(CTest)
    add_executable(service_starter_lifecycle_tests tests/lifecycle_tests.cpp)
    target_link_libraries(service_starter_lifecycle_tests PRIVATE service_starter::service_starter)
    service_starter_configure(service_starter_lifecycle_tests)
    add_test(NAME service_starter_lifecycle COMMAND service_starter_lifecycle_tests)
    set_tests_properties(service_starter_lifecycle PROPERTIES TIMEOUT 60)
    if(WIN32)
        add_executable(service_starter_scm_tests tests/windows_scm_tests.cpp)
        target_link_libraries(service_starter_scm_tests PRIVATE service_starter::service_starter)
        target_compile_definitions(service_starter_scm_tests PRIVATE UNICODE _UNICODE)
        service_starter_configure(service_starter_scm_tests)
        add_test(NAME service_starter_scm COMMAND service_starter_scm_tests)
        set_tests_properties(service_starter_scm PROPERTIES TIMEOUT 60)
    endif()
endif()
